# TRABALHANDO COM ARQUIVOS

touch nome-arquivo
- cria um arquivo
  
mv nome-arquivo meu-diretorio
- move o arquivo informado pro diretorio informado

mv nome-atual nome-novo.txt
- renomeia o nome do arquivo pra um novo nome

cp diretori/arquivo.txt
- copia o arquivo pro diretorio atual

ls -a
- exibe arquivos ocultos

ll
- exibe detalhadamente os arquivos ocultos

echo seu-texto > name-arquivo.txt
- escreve o texto informado no arquivo informado
  
cat name-arquivo.txt
- le o arquivo
  
file name-arquivo.txt
- exibe infos sobre o arquivo

rm name-arquivo.txt
- remove o arquivo

# TRABALHANDO COM DIRETÓRIOS

cd /
- navega até o diretório raiz (root)

ls
- lista os arquivos e diretórios de onde estamos

cd ..
- navega pro diretório anterior de onde estamos

cd ~
- navega até home (diretório do usuário)

mkdir name-diretorio
- cria um diretório com o nome informado

file name-diretorio
- exibe infos sobre o diretorio

rmdir
- remove diretorio se estive vazio

rm -r name-diretorio
- remove diretorio com conteudo
